\section{Motivation}
% =============================================================================

\begin{frame}{The benchmark picture looks solved}
  \begin{itemize}
    \item Modern end-to-end planners report strong aggregate PDMS on NAVSIM.
    \item Top methods sit in the $89$--$94$ PDMS range; TransFuser at $83.9$, DiffusionDrive at $88.1$.
    \item The headline metric hides a structural weakness: \textbf{train and test cities are pooled}.
  \end{itemize}
  \vspace{4pt}
  \begin{block}{What gets measured}
    Pooled NAVSIM evaluates \emph{interpolation} inside a known mix of Boston, Pittsburgh, Las Vegas, and Singapore --- not transfer to a new city.
  \end{block}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Geographic shift breaks this picture}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \textbf{Open-loop, nuScenes, Boston $\to$ Singapore:}
    \begin{itemize}
      \item Supervised Swin: L2 inflates \alert{$9.77\times$}, collisions \alert{$19.34\times$}.
      \item Domain-pretrained I-JEPA (rect, frozen): L2 \textcolor{ssl}{$1.20\times$}, collisions \textcolor{ssl}{$0.75\times$}.
    \end{itemize}
    \vspace{6pt}
    \textbf{Closed-loop, NAVSIM, Singapore-trained:}
    \begin{itemize}
      \item ResNet-34 Latent TransFuser: $48.9\%$ average OOD PDMS.
      \item Singapore uses left-hand traffic; transfer is asymmetric.
    \end{itemize}
    \column{0.45\textwidth}
    \begin{block}{Observation}
      Two models with similar pooled PDMS can differ by an \emph{order of magnitude} under explicit city transfer.
      \smallskip
      \\{\footnotesize Source: Naeinian et al., prior cross-city benchmark study.}
    \end{block}
  \end{columns}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Central claim of the thesis}
  \begin{block}{}
    \large
    \textbf{Cross-city generalization failure in end-to-end driving is primarily a \emph{representation} problem, not a planning-architecture problem.}
    \smallskip
    \\Self-supervised pretraining produces features that are inherently more transferable across cities, \emph{reducing generalization gaps by an order of magnitude}.
  \end{block}
  \vspace{8pt}
  Operationalized as six research questions, covered across four empirical phases.
\end{frame}

\section{Background}
% =============================================================================

\begin{frame}{NAVSIM in one slide}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \textbf{Inputs.} 8 surround cameras, merged LiDAR, ego state, route.
    \medskip
    \\\textbf{Output.} 8 future poses $(x, y, \psi)$ over 4\,s at 0.5\,s intervals.
    \medskip
    \\\textbf{Scorer.} Predictive Driver Model Score:
    \[
      \mathrm{PDMS} = \mathrm{NC}\cdot\mathrm{DAC}\cdot\tfrac{5\,\mathrm{EP} + 5\,\mathrm{TTC} + 2\,\mathrm{C}}{12}
    \]
    {\footnotesize NC / DAC enter multiplicatively --- one violation zeroes the scenario.}
    \column{0.45\textwidth}
    \textbf{Four cities (right-hand):}
    \begin{itemize}\footnotesize
      \item Las Vegas: $818$ train / $50$ test
      \item Boston: $255$ train / $62$ test
      \item Pittsburgh: $140$ train / $22$ test
    \end{itemize}
    \textbf{One left-hand:}
    \begin{itemize}\footnotesize
      \item Singapore: $97$ train / $13$ test
    \end{itemize}
    \smallskip
    Default NAVSIM pools all four.
  \end{columns}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Three SSL paradigms studied}
  \begin{columns}[T]
    \column{0.33\textwidth}
    \textbf{I-JEPA}\\{\footnotesize Assran et al., 2023}
    \begin{itemize}\footnotesize
      \item Predict \emph{latents} of masked target regions from visible context.
      \item Loss in representation space, stop-grad EMA target encoder.
      \item No pixel reconstruction, no contrastive negatives.
    \end{itemize}
    \column{0.33\textwidth}
    \textbf{DINOv2}\\{\footnotesize Oquab et al., 2024}
    \begin{itemize}\footnotesize
      \item Student/teacher \emph{self-distillation}.
      \item Multi-crop views, prototype softmax match.
      \item Scales on LVD-142M.
    \end{itemize}
    \column{0.33\textwidth}
    \textbf{MAE}\\{\footnotesize He et al., 2022}
    \begin{itemize}\footnotesize
      \item \emph{Pixel reconstruction} from $75\%$-masked patches.
      \item Asymmetric encoder/decoder; decoder discarded.
      \item No teacher.
    \end{itemize}
  \end{columns}
  \vspace{8pt}
  \begin{block}{Why study all three}
    Different invariances. I-JEPA biases to semantic layout; DINOv2 to augmentation-invariant identity; MAE to pixel-level appearance. Geographic transfer punishes the wrong invariance set.
  \end{block}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Planning architectures evaluated}
  \begin{columns}[T]
    \column{0.24\textwidth}
    \textbf{LAW}
    \smallskip
    \\\footnotesize Open-loop, nuScenes. Swin-T baseline. World-model auxiliary loss.
    \column{0.24\textwidth}
    \textbf{TransFuser}
    \smallskip
    \\\footnotesize Closed-loop, NAVSIM. ResNet-34 + LiDAR fusion.
    \column{0.24\textwidth}
    \textbf{Latent TransFuser}
    \smallskip
    \\\footnotesize TransFuser with the LiDAR branch replaced by a learned latent --- camera-only probe.
    \column{0.24\textwidth}
    \textbf{DiffusionDrive}
    \smallskip
    \\\footnotesize Truncated 2-step diffusion from K-means anchor trajectories; 45 FPS.
  \end{columns}
  \vspace{10pt}
  \begin{block}{Design principle}
    Every planner is held fixed; \emph{only the image backbone} is varied. Any downstream change is attributable to the representation, not to the head.
  \end{block}
\end{frame}

\section{Methodology}

\begin{frame}{NAVSIM City Splits --- benchmark contribution}
  \begin{itemize}
    \item NAVSIM exposes no per-city partition; city labels are recoverable from nuPlan log prefixes (\texttt{us-ma-boston}, \texttt{sg-one-north}, \dots).
    \item A single preprocessing pass assigns every scenario token to exactly one city; JSON token lists feed the dataloader.
    \item Turns NAVSIM from a pooled benchmark into a \textbf{reproducible geographic-transfer benchmark}.
  \end{itemize}

% -----------------------------------------------------------------------------
\begin{frame}{SSL pre-training: Fatemeh Naeinian's protocol}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \textbf{Shared substrate}
    \begin{itemize}\footnotesize
      \item nuScenes, \emph{all six cameras}, $168{,}780$ frames.
      \item ViT-S/14 for every SSL family.
      \item Two resolutions: $224{\times}224$ (square) and $224{\times}392$ (rectangular, native aspect).
      \item No photometric augmentation --- the objective, not the pretext, is the variable.
      \item 100 epochs, AdamW, cosine schedule. $1{\times}$ H200.
    \end{itemize}
    \column{0.45\textwidth}
    \textbf{Per-family knobs}
    \begin{itemize}\footnotesize
      \item I-JEPA: 1 context block $85$--$100\%$, 4 targets $15$--$20\%$; EMA $0.996\!\to\!1.0$.
      \item DINOv2: 2 global + 6 local crops; teacher momentum $0.9985$.
      \item MAE: $75\%$ random mask; decoder $(8, 512, 16)$.
    \end{itemize}
  \end{columns}
  \vspace{6pt}
  \begin{block}{Credit}
    All domain-specific SSL pretraining in this thesis was performed by \textbf{Fatemeh Naeinian}, first author of our earlier cross-city benchmark study. This thesis consumes her checkpoints through a shared backbone adapter.
  \end{block}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Generalization-gap metrics}
  \textbf{Open-loop error ratio} (lower is better):
  \[
    R_\text{err}(c_\text{train} \to c_\text{test})
    \;=\;
    \frac{E_\text{cross}(c_\text{train} \to c_\text{test})}
         {E_\text{in}(c_\text{train} \to c_\text{train})}
  \]
  $R_\text{err}{=}1$ is perfect preservation, $R_\text{err}{>}1$ is degradation.
  \medskip
  \\
  \textbf{Closed-loop OOD PDMS} (higher is better):
  \[
    S_\text{OOD}(c_i) \;=\; \tfrac{1}{|\mathcal{C}|-1}\sum_{c_j \neq c_i} S(c_i \to c_j)
  \]
  \smallskip
  Both follow Naeinian et al., prior cross-city benchmark study. Ratios are reported \emph{alongside} the full matrix to flag degenerate in-domain denominators.
\end{frame}

\section{Exploratory Lightweight-Head Study}
% =============================================================================

\begin{frame}{Exploratory study: minimum planner, varied backbone}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{\texttt{LightweightPlanningAgent}}
    \begin{itemize}\footnotesize
      \item One front camera, $224{\times}224$.
      \item 2-layer MLP head ($1288\!\to\!256\!\to\!256\!\to\!T{\cdot}3$).
      \item $L_1$ loss only; no world-model, no perception auxiliaries.
      \item AdamW, 20 epochs, cosine, frozen / fully unfrozen only.
    \end{itemize}
    \column{0.48\textwidth}
    \textbf{45-run experiment matrix}
    \begin{itemize}\footnotesize
      \item[$\bullet$] P1: 10 backbones $\times$ frozen
      \item[$\bullet$] P2: 16 cross-city + 4 fusion variants
      \item[$\bullet$] P3: 10 backbones $\times$ fully unfrozen
      \item[$\bullet$] P4: 5 multi-cam / cross-attn / Tx head
    \end{itemize}
    \smallskip
    Driven by one \texttt{LightweightHeadConfig}.
  \end{columns}
  \vspace{6pt}
  \begin{block}{Purpose}
    Diagnostic, not competitive. With planner capacity pinned at its minimum, any PDMS spread \emph{is} a representation spread.
  \end{block}
\end{frame}

\section{Cross-City SSL Benchmark}
% =============================================================================

\begin{frame}{Cross-city benchmark: strict city-disjoint splits}
  \begin{columns}[T]
    \column{0.46\textwidth}
    \textbf{Open-loop, nuScenes (LAW)}
    \begin{itemize}\footnotesize
      \item Bidirectional Boston $\leftrightarrow$ Singapore.
      \item 12 ep, AdamW $5\!\times\!10^{-5}$, cosine, L40S, batch 2.
      \item \emph{Implemented and run by Fatemeh Naeinian.}
    \end{itemize}
    \medskip
    \textbf{Closed-loop, NAVSIM (TF / LatTF)}
    \begin{itemize}\footnotesize
      \item Full $4{\times}4$ train / test city matrix.
      \item 20 ep, plain Adam $10^{-4}$, no wd / no scheduler, 4$\times$H200, effective batch 128.
      \item ViT-into-CNN adapter + config hardening --- author's contribution.
    \end{itemize}
    \column{0.48\textwidth}
    \textbf{11-backbone matrix per leg}
    \begin{itemize}\footnotesize
      \item Supervised defaults: Swin-T / ResNet-34.
      \item Large-scale SSL: I-JEPA ViT-H/14, DINOv2 ViT-S/14, MAE ViT-B/16.
      \item Capacity-matched domain SSL: 3 objectives $\times$ 2 resolutions on ViT-S/14.
      \item Each row $\times$ \{frozen, fully unfrozen\}.
    \end{itemize}
  \end{columns}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Transfer is structurally asymmetric}
  \textbf{Boston $\to$ Singapore is harder than Singapore $\to$ Boston} for every backbone.
  \medskip
  \begin{itemize}
    \item Three of four cities are right-hand traffic (Boston, Pittsburgh, Las Vegas).
    \item Singapore: left-hand traffic + smallest data pool + distinct road paint.
    \item Models trained on RHT data overfit lane-geometry priors that do not exist in Singapore.
  \end{itemize}
  \vspace{6pt}
  \begin{block}{Supervised vs.\ SSL}
    Supervised ResNet-34 shows the largest asymmetry. Frozen SSL backbones collapse the gap --- the SSL feature space is closer to \emph{city-invariant} by construction.
  \end{block}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Cross-city benchmark --- takeaways}
  \begin{enumerate}
    \item \textbf{Representation dominates planner} under explicit geographic shift.
    \item \textbf{Domain-specific nuScenes pretraining} reduces the generalization gap roughly $8\times$ over supervised Swin on the hardest open-loop transfer.
    \item \textbf{Singapore is the stress-test city}; SSL narrows but does not eliminate the directional asymmetry.
    \item \textbf{Removing LiDAR amplifies the SSL advantage}, because the planner can no longer route around a weak image representation.
  \end{enumerate}
\end{frame}

\section{SSL Backbones in Generative Planning}
% =============================================================================

\begin{frame}{Why DiffusionDrive --- a hostile-to-backbone planner}
  \textbf{DiffusionDrive attenuates the backbone by design.}
  \begin{itemize}\footnotesize
    \item \textbf{K-means anchors.} Denoising starts from one of 256 expert-trajectory prototypes, not noise. Weak features can still exit near a valid plan.
    \item \textbf{Truncated 2-step denoising.} Only two conditioning rounds at inference; each contributes less than it would under a 50-step denoiser.
    \item \textbf{Grid-sampled BEV cross-attention.} Quantizes backbone access to waypoint-local grid samples, not a dense feature map.
    \item \textbf{Time-only FiLM.} The only gate that fires is on the denoising timestep, not on the visual features.
  \end{itemize}
  \begin{block}{Evaluation criterion}
    Under a head that absorbs backbone differences, \emph{any surviving cross-city SSL advantage is stronger evidence than a large advantage under a weak planner}. We therefore report \textbf{transfer-ratio inflation}, not absolute PDMS. Adapter is identical to the cross-city benchmark's (same \texttt{create\_image\_encoder} factory) --- the head is the only manipulated variable.
  \end{block}
\end{frame}

\section{Mechanistic Analysis of Representations}
% =============================================================================

\begin{frame}{Why does SSL transfer better?}
  \textbf{Five complementary probes} applied to the cross-city Tier-A checkpoints:
  \begin{enumerate}
    \item \textbf{CKA} across cities --- does the feature covariance structure align?
    \item \textbf{City-classification linear probing} --- how much city identity leaks into the representation?
    \item \textbf{MMD} between per-city feature distributions --- distance in RKHS.
    \item \textbf{Effective rank} --- does SSL maintain higher-dimensional subspaces?
    \item \textbf{Attention-relevance maps} --- which pixels drive the planner under city shift?
  \end{enumerate}
  \medskip
  \begin{block}{Expected pattern}
    SSL: higher cross-city CKA, lower linear-probe city accuracy, smaller MMD, higher effective rank. Supervised ResNet-34: the reverse. No prior end-to-end driving study combines all five under a strict geographic holdout.
  \end{block}
\end{frame}

\section{Contributions and Future Work}
% =============================================================================

\begin{frame}{Contributions}
  \begin{enumerate}
    \item \textbf{NAVSIM City Splits benchmark.} First per-city train/test partitioning of NAVSIM across four cities. The protocol the rest of the thesis is built on.
    \item \textbf{Cross-city SSL benchmark.} 11 backbones $\times$ 2 adaptation modes $\times$ 4 cities on LAW, TransFuser, Latent TransFuser. Supervised $\to$ SSL reduces open-loop transfer inflation by an order of magnitude.
    \item \textbf{Exploratory lightweight-head agent.} Backbone-agnostic \texttt{LightweightPlanningAgent} with 45 runs pinned at minimum planner capacity; isolates representation from planner.
    \item \textbf{DiffusionDrive + SSL integration.} First camera-only SSL + truncated-diffusion planner on NAVSIM; matches camera+LiDAR ResNet within $\approx 1$ PDMS.
    \item \textbf{Mechanistic analysis framework.} CKA, linear probing, MMD, effective rank, and attention-relevance applied to driving backbones under strict city holdout.
  \end{enumerate}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Limitations}
  \begin{itemize}
    \item Open-loop evaluation covers two cities (nuScenes); closed-loop four (NAVSIM). Broader geographies would strengthen external validity.
    \item Weather, lighting, and seasonal effects are not disentangled from geographic shift.
    \item Single training runs per cell; variance across seeds is not measured.
    \item Domain-specific pretraining is on nuScenes only; extending to \texttt{navtrain} or video SSL (V-JEPA) is future work.
    \item RL post-training (GRPO, DPO) is out of scope for this thesis.
  \end{itemize}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Future work}
  \begin{itemize}
    \item \textbf{Video SSL.} V-JEPA and Drive-JEPA-style video pretraining for the same cross-city protocol.
    \item \textbf{RL transfer.} Does GRPO / DPO fine-tuning of DiffusionDrive survive geographic transfer?
    \item \textbf{Weather / lighting factorization.} Joint city $\times$ weather splits to isolate geography from appearance.
    \item \textbf{More cities.} Beyond nuScenes + NAVSIM, extend to nuPlan's extended logs and to non-US/SG datasets.
    \item \textbf{Mechanistic $\to$ training-time feedback.} Use CKA / MMD as pretraining auxiliaries.
  \end{itemize}
\end{frame}
